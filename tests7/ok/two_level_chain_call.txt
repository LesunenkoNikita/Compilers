struct Point
{
    i32 mut x
    i32 mut y

    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }

    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
}
struct Circle
{
    Point center
    i32 radius

    fn area2 := () -> i32
    {
        i32 a{radius * radius * 3}
        exit a
    }
}
Point p{6, 8}
Circle c{p, 1}
exit c.center.d2()
