struct Point
{
    i32 mut x
    i32 mut y
    fn moved := (i32 dx) -> Point
    {
        Point q{x + dx, y}
        exit q
    }
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
}
Point p{1, 2}
print(p.moved(1).moved(2))
print(p.moved(5).shift(5))
print(p)
exit p.moved(10).moved(20).x
